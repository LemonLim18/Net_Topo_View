! --- Management Interface (→ Cloud0 → 192.168.1.0/24 real LAN) ---
!
! Rule: put the management IP on the Cloud0-connected interface.
! For ROUTERS (CSR, IOL L3): IP goes directly on the physical port — no VLAN.
! For L2 SWITCHES (IOL L2): physical port has NO IP (it's a switchport);
!   IP goes on the SVI (interface Vlan100) + set ip default-gateway.
!
! ── BLOCK A: CSR 1000v (CORE-R1, CORE-R2) ──────────────────
! Cloud0 port = whichever GiX you wired Cloud0 to in EVE-NG.
! Example: if Cloud0 is on the 6th interface slot → GigabitEthernet6.
interface GigabitEthernet{{CLOUD0_GI_NUM}}
 description MGMT-CLOUD0-REAL-LAN
 ip address {{MGMT_IP}} 255.255.255.0
 no ip proxy-arp
 negotiation auto
 no mop enabled
 no mop sysid
 no shutdown
!
! ── BLOCK B: IOL L3 (DIST-SW1–SW4) — use instead of Block A ─
! e0/0 defaults to a ROUTED port on IOL L3, so IP directly on it is fine.
! interface Ethernet0/0
!  description MGMT-CLOUD0-REAL-LAN
!  ip address {{MGMT_IP}} 255.255.255.0
!  no ip proxy-arp
!  no shutdown
!
! ── BLOCK C: IOL L2 (ACC-SW1–SW6) — use instead of Block A ──
! ALL ports on IOL L2 are switchports. You CANNOT put an IP on e0/0.
! Instead: set e0/0 as access port for VLAN 100, put the IP on the SVI.
! interface Ethernet0/0
!  description MGMT-CLOUD0-ACCESS-VLAN100
!  switchport mode access
!  switchport access vlan 100
!  no shutdown
! !
! interface Vlan100
!  description MGMT-SVI
!  ip address {{MGMT_IP}} 255.255.255.0
!  no shutdown
! !
! ip default-gateway 192.168.1.1   ! your LAN gateway — enables SSH/SNMP reach from outside
!

! --- Loopback0 / OSPF Router-ID (Core + Distribution) ---
!
! You CANNOT reuse the management IP on Loopback0 — IOS rejects the overlap.
! Pick ONE of these two approaches:
!
! OPTION A (recommended — no loopback needed):
!   Just set router-id explicitly under OSPF:
!     router ospf 1
!      router-id {{MGMT_IP}}
!
! OPTION B (separate loopback — for BGP/MPLS source later):
!   Use a dedicated range like 10.255.0.x/32:
!   interface Loopback0
!    description ROUTER-ID
!    ip address 10.255.0.{{LAST_OCTET}} 255.255.255.255
!    no shutdown
!   Then reference it: router-id 10.255.0.{{LAST_OCTET}}
!
